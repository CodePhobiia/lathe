    lathe.executor ¶

    Build executor: orchestrates every stage in STAGE_ORDER.

    Stage 2.IV (Wave 2) decomposed this module from a 2,315-LOC god-class into a thin dispatcher that delegates per-stage step bodies to lathe.core.stages.*. The public surface — BuildError, build_asset, build_surface_stage, validate_surface_stage, repair_surface_stage, BuildExecutor, RuntimeMesh, STAGE_ORDER — is preserved unchanged. Implementation helpers (_run_reference_topology_cleanup) are also re-exported from the extracted lathe.core.geometry module to keep existing imports working.

    Do NOT change the stable contract lathe.shape.intent_mesh_handoff_v1 emitted by the shape stage. See lathe/shape/ for the contract bytes.

    def build(self) -> dict[str, Any]:
    """Build.

    Returns:
        The result.
    """
    diagnostics = check_program(self.program)
    errors = [item for item in diagnostics if item["severity"] == "error"]
    if errors:
        raise BuildError("Build blocked by semantic errors", diagnostics=diagnostics)

    cached = self._restore_cached_build()
    if cached is not None:
        return cached

    self._prepare_dirs()
    dump_json(self.normalized_asset_path, self.normalized_program)
    dump_json(self.out_dir / "plan.json", self.plan)
    dump_json(self.out_dir / "materialized_asset.json", self.asset)

    # Wall-clock budget for the build. Default 5 min for the conservative
    # path; override via runtime_config.execution.timeout_s. The marquee
    # courier_drone build was previously hanging silently past 15 min;
    # this turns "silent hang" into a structured BuildError with
    # ``stage_timeout`` diagnostics so the user knows which stage stalled.
    execution_cfg = self.runtime_config.get("execution") or {}
    timeout_s = float(execution_cfg.get("timeout_s") or DEFAULT_BUILD_TIMEOUT_S)
    build_start = time.monotonic()

    def _check_budget(stage_name: str) -> None:
        elapsed = time.monotonic() - build_start
        if elapsed > timeout_s:
            raise BuildError(
                f"build wall-clock budget exceeded after {elapsed:.0f}s "
                f"(limit={timeout_s:.0f}s, last stage={stage_name})",
                diagnostics=[
                    {
                        "code": "stage_timeout",
                        "severity": "error",
                        "message": (
                            f"Stage '{stage_name}' completed but the cumulative "
                            f"build wall-clock exceeded the configured budget of "
                            f"{timeout_s:.0f}s. Override via "
                            "runtime_config.execution.timeout_s."
                        ),
                        "phase": stage_name,
                    }
                ],
            )

    self._publish_pre_shape_quality_context()
    shape_bundle = stage_shape.run(self)
    _check_budget("shape")
    topology_bundle = stage_topology.run(self, shape_bundle)
    _check_budget("topology")
    uv_bundle = stage_uv.run(self, topology_bundle)
    _check_budget("uv")
    surface_bundle = stage_surface.run(self, uv_bundle)
    _check_budget("surface")
    rig_bundle = stage_rig.run(self, surface_bundle)
    _check_budget("rig")
    interfaces_bundle = stage_interfaces.run(self, rig_bundle)
    _check_budget("interfaces")
    physics_bundle = stage_physics.run(self, interfaces_bundle)
    _check_budget("physics")
    optimize_bundle = stage_optimize.run(self, physics_bundle)
    _check_budget("optimize")
    stage_execution.run(self, optimize_bundle)
    _check_budget("execution")
    # Phase 10: publish a redacted snapshot of the runtime config so the
    # vision-critique validator can read the optional ``vision_critique``
    # provider block without ever seeing raw api keys / endpoints.
    self.runtime["_runtime_config"] = redact_secrets(self.runtime_config)
    # Phase 11: publish the reference / moodboard fixtures so the
    # validator can build the additive ``reference_board`` payload and
    # the design-brief layer can merge reference negatives.
    # Phase 13: honor the optional experimental override path.
    self._publish_reference_board_context()
    # Phase 12: publish the standard hard-surface kit library so the
    # validator can build the additive ``kit_library`` payload.
    self.runtime["_kit_library"] = _safe_load_kit_library()
    validate_bundle = stage_validate.run(self, optimize_bundle)
    repair_rounds: list[dict[str, Any]] = []
    if not validate_bundle["certified"]:
        repair_cfg = self.runtime_config.get("repair") or {}
        if repair_cfg.get("autonomous", True):
            from .repair_loop import run_repair_loop

            # Precedence: ``repair.max_rounds_override`` in runtime_config
            # (invocation-time override, opt-in via CLI / runtime config)
            # wins over ``asset.validate.repair.max_rounds``, which wins
            # over the runtime_config default. The asset value stays
            # authoritative for production; the override surface lets
            # iteration runs cap the repair loop without editing sources.
            asset_repair_cfg = (
                (self.asset.get("validate", {}) or {}).get("repair", {}) or {}
            )
            asset_max_rounds = asset_repair_cfg.get("max_rounds")
            override_max_rounds = repair_cfg.get("max_rounds_override")
            if override_max_rounds is not None:
                try:
                    effective_max_rounds = int(override_max_rounds)
                except (TypeError, ValueError):
                    effective_max_rounds = (
                        int(asset_max_rounds) if asset_max_rounds is not None
                        else int(repair_cfg.get("max_rounds", 3))
                    )
            elif asset_max_rounds is not None:
                try:
                    effective_max_rounds = int(asset_max_rounds)
                except (TypeError, ValueError):
                    effective_max_rounds = int(repair_cfg.get("max_rounds", 3))
            else:
                effective_max_rounds = int(repair_cfg.get("max_rounds", 3))
            validate_bundle, repair_rounds = run_repair_loop(
                self,
                validate_bundle,
                max_rounds=effective_max_rounds,
                min_improvement=float(repair_cfg.get("min_improvement", 0.01)),
                budget_seconds=float(repair_cfg.get("budget_seconds", 180.0)),
            )
            # Refresh optimize/execution bundles in case a round regenerated the stack.
            optimize_bundle = self.runtime.get("optimize", {}).get("bundle", optimize_bundle)
            stage_execution.run(self, optimize_bundle)
    if not validate_bundle["certified"]:
        raise BuildError("Validation failed", diagnostics=validate_bundle["diagnostics"])
    package_bundle = stage_package.run(self, optimize_bundle, validate_bundle)

    if "kind" not in self.asset:
        raise BuildError(
            f"materialized asset {self.asset_name!r} is missing required "
            "'kind' field; resolver/normalizer mismatch"
        )
    report = {
        "asset": self.asset_name,
        "kind": self.asset["kind"],
        "out_dir": str(self.out_dir),
        "certified": True,
        "backend_stack": self.backend_stack,
        "selected_backends": self.selected_backends,
        "runtime_config": redact_secrets(self.runtime_config),
        "stage_bundles": {
            "shape": relpath(self.out_dir, self.stage_dirs["shape"] / "bundle.json"),
            "topology": relpath(self.out_dir, self.stage_dirs["topology"] / "bundle.json"),
            "uv": relpath(self.out_dir, self.stage_dirs["uv"] / "bundle.json"),
            "surface": relpath(self.out_dir, self.stage_dirs["surface"] / "bundle.json"),
            "execution": relpath(self.out_dir, self.stage_dirs["execution"] / "bundle.json"),
            "rig": relpath(self.out_dir, self.stage_dirs["rig"] / "bundle.json"),
            "interfaces": relpath(self.out_dir, self.stage_dirs["interfaces"] / "bundle.json"),
            "physics": relpath(self.out_dir, self.stage_dirs["physics"] / "bundle.json"),
            "optimize": relpath(self.out_dir, self.stage_dirs["optimize"] / "bundle.json"),
            "validate": relpath(self.out_dir, self.stage_dirs["validate"] / "report.json"),
            "package": relpath(self.out_dir, self.stage_dirs["package"] / "bundle.json"),
        },
        "outputs": package_bundle["outputs"],
        "manifest_path": package_bundle["manifest_path"],
        "trace_path": package_bundle.get("trace_path"),
        "score": validate_bundle["score"],
        "threshold": validate_bundle["threshold"],
        "repair_rounds": repair_rounds,
        "optimize": {
            "lod_manifest_path": optimize_bundle.get("lod_manifest_path"),
            "lods": optimize_bundle.get("lods"),
            "collision": optimize_bundle.get("collision"),
        },
        # B9 — surface texel density evidence at the build_report
        # top level. Authored vs actual, worst-chart drift, and
        # chart count are read from the uv bundle. Without this
        # the texel-density audit is buried in stage bundles and
        # invisible to anyone judging the build at a glance.
        "uv_evidence": _uv_evidence_summary(self.runtime.get("uv", {})),
        # M3 — surface ZoneAtlas evidence at the build_report
        # top level. Reads from the topology bundle's zone_atlas
        # (populated by the procedural_program_path lowering).
        "zone_atlas_summary": _zone_atlas_summary(
            self.runtime.get("topology", {})
        ),
        "cache": {
            "hit": False,
            "persisted": False,
            "reason": "pending_auto_persist",
        },
    }
    dump_json(self.out_dir / "build_report.json", report)
    cache_status = self._persist_build_cache()
    report["cache"] = cache_status
    dump_json(self.out_dir / "build_report.json", report)
    if cache_status.get("persisted"):
        with contextlib.suppress(OSError):
            dump_json(self.cache_dir / "build_report.json", report)
    return report

    def quick_build(self, *, glb_path: Path, use_cache: bool = True) -> dict[str, Any]:
    """Run the minimal preview pipeline: shape -> topology -> uv -> surface (no bakes) -> GLB.

    Skips execution, rig, interfaces, physics, optimize, validate, package,
    and repair. Targets <60s wall-clock on hero assets vs. ~25 min for the
    full ``build`` pipeline. Used by ``lathe quick``.

    Returns a small report dict with asset, out_dir, glb path, and per-stage
    wall-clock timings. When ``use_cache`` is true, attempts to restore the
    previously-cached out_dir + GLB before running any stages — this turns
    warm iteration loops on unchanged sources into <5s file copies.
    """
    from .core.io import export_runtime_scene_glb

    glb_path = Path(glb_path)
    if use_cache:
        cached = self._restore_quick_cache(glb_path)
        if cached is not None:
            return cached

    self._prepare_uncached_build()
    self._publish_pre_shape_quality_context()
    timings: dict[str, float] = {}

    t0 = time.monotonic()
    shape_bundle = stage_shape.run(self)
    timings["shape"] = time.monotonic() - t0

    t0 = time.monotonic()
    topology_bundle = stage_topology.run(self, shape_bundle)
    timings["topology"] = time.monotonic() - t0

    t0 = time.monotonic()
    uv_bundle = stage_uv.run(self, topology_bundle)
    timings["uv"] = time.monotonic() - t0

    t0 = time.monotonic()
    stage_surface.run(self, uv_bundle, skip_bakes=True)
    timings["surface"] = time.monotonic() - t0

    t0 = time.monotonic()
    uv_runtime_mesh: RuntimeMesh = self.runtime["uv"]["runtime_mesh"]
    glb_path.parent.mkdir(parents=True, exist_ok=True)
    export_runtime_scene_glb(uv_runtime_mesh, glb_path)
    timings["package"] = time.monotonic() - t0

    face_parts = list(getattr(uv_runtime_mesh, "face_parts", []) or [])
    part_count = len({p for p in face_parts if p}) or 1
    report = {
        "asset": self.asset_name,
        "out_dir": str(self.out_dir),
        "glb_path": str(glb_path),
        "part_count": part_count,
        "face_count": len(uv_runtime_mesh.mesh.faces),
        "vertex_count": len(uv_runtime_mesh.mesh.vertices),
        "stage_timings_s": {k: round(v, 3) for k, v in timings.items()},
        "cache_hit": False,
        "stage_bundles": {
            "shape": relpath(self.out_dir, self.stage_dirs["shape"] / "bundle.json"),
            "topology": relpath(self.out_dir, self.stage_dirs["topology"] / "bundle.json"),
            "uv": relpath(self.out_dir, self.stage_dirs["uv"] / "bundle.json"),
            "surface": relpath(self.out_dir, self.stage_dirs["surface"] / "bundle.json"),
        },
    }
    if use_cache:
        self._persist_quick_cache(glb_path, report)
    return report

    initial = self.validate_surface_stage()

    def clear_quick_cache() -> int:
    """Remove every cached ``lathe quick`` entry. Returns the count removed."""
    root = _quick_cache_root()
    if not root.exists():
        return 0
    removed = 0
    for child in root.iterdir():
        if child.is_dir():
            shutil.rmtree(child, ignore_errors=True)
            removed += 1
    return removed
